    <script src="game.js"></script>

    <script>

        const canvas = document.getElementById('gameCanvas');

        const ctx = canvas.getContext('2d');

        canvas.width = canvas.offsetWidth;

        canvas.height = canvas.offsetHeight;

        

        let ball = { 

            x: canvas.width / 2, 

            y: canvas.height / 2, 

            dx: 4, 

            dy: 3, 

            size: 10 

        };

        let catPaddle = { x: 60, y: canvas.height / 2 - 30, width: 15, height: 60 };

        let handPaddle = { x: canvas.width - 80, y: canvas.height / 2 - 30, width: 15, height: 60 };

        let score = 0;

        

        document.getElementById('instructions').style.display = 'block';

        setTimeout(() => {

            document.getElementById('instructions').style.display = 'none';

        }, 4000);

        

        canvas.addEventListener('mousemove', (e) => {

            const rect = canvas.getBoundingClientRect();

            const y = e.clientY - rect.top;

            catPaddle.y = Math.max(0, Math.min(canvas.height - catPaddle.height, y - catPaddle.height / 2));

        });

        

        canvas.addEventListener('touchmove', (e) => {

            e.preventDefault();

            const rect = canvas.getBoundingClientRect();

            const y = e.touches[0].clientY - rect.top;

            catPaddle.y = Math.max(0, Math.min(canvas.height - catPaddle.height, y - catPaddle.height / 2));

        }, { passive: false });

        

        function animate() {

            ctx.clearRect(0, 0, canvas.width, canvas.height);

            

            ctx.fillStyle = '#87CEEB';

            ctx.fillRect(0, 0, canvas.width, canvas.height);

            

            ball.x += ball.dx;

            ball.y += ball.dy;

            

            if (ball.y - ball.size < 0 || ball.y + ball.size > canvas.height) {

                ball.dy = -ball.dy;

            }

            

            if (ball.x - ball.size < catPaddle.x + catPaddle.width &&

                ball.x + ball.size > catPaddle.x &&

                ball.y > catPaddle.y &&

                ball.y < catPaddle.y + catPaddle.height) {

                ball.dx = Math.abs(ball.dx) + 0.3;

                score++;

            }

            

            const handCenter = handPaddle.y + handPaddle.height / 2;

            if (handCenter < ball.y - 20) handPaddle.y += 2.5;

            else if (handCenter > ball.y + 20) handPaddle.y -= 2.5;

            handPaddle.y = Math.max(0, Math.min(canvas.height - handPaddle.height, handPaddle.y));

            

            if (ball.x + ball.size > handPaddle.x &&

                ball.x - ball.size < handPaddle.x + handPaddle.width &&

                ball.y > handPaddle.y &&

                ball.y < handPaddle.y + handPaddle.height) {

                ball.dx = -Math.abs(ball.dx) - 0.3;

            }

            

            if (ball.x < 0) {

                catState.happiness = Math.max(0, catState.happiness - 5);

                saveState();

                window.location.href = 'index.html';

            }

            if (ball.x > canvas.width) {

                ball.x = canvas.width / 2;

                ball.y = canvas.height / 2;

                ball.dx = -4;

                score++;

                if (score >= 5) {

                    catState.happiness = Math.min(100, catState.happiness + 15);

                    saveState();

                    window.location.href = 'index.html';

                }

            }

            

            ctx.fillStyle = '#C0C0C0';

            ctx.fillRect(catPaddle.x, catPaddle.y, catPaddle.width, catPaddle.height);

            

            ctx.fillStyle = '#FFB6C1';

            ctx.fillRect(handPaddle.x, handPaddle.y, handPaddle.width, handPaddle.height);

            

            ctx.fillStyle = '#FFFF00';

            ctx.beginPath();

            ctx.arc(ball.x, ball.y, ball.size, 0, Math.PI * 2);

            ctx.fill();

            

            ctx.fillStyle = '#FFFF00';

            ctx.font = '10px "Press Start 2P"';

            ctx.fillText('СЧЁТ: ' + score + '/5', canvas.width / 2 - 50, 30);

            

            requestAnimationFrame(animate);

        }

        

        animate();

    </script>
